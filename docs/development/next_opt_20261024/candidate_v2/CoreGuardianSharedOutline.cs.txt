using System.Collections.Generic;
using UnityEngine;
using UnityEngine.Rendering;
using Unity.Profiling;

// Default ON since 2026-10-24 (outline redesign plan, option A; see docs/development/shared_outline_20261024).
// All always-on emphasis bodies write their entity id into ONE screen-size mask, drawn in visual
// sorting order so each pixel keeps the front-most body. Each body keeps its own output quad in the
// original sorting slot (front part +3), so overlap order with other bodies/HUD/effects is unchanged.
// Selection (HUD) and upgrade overlays keep the original per-body CoreGuardianSelectionOutline path.
internal sealed class CoreGuardianSharedOutline
{
    static readonly ProfilerMarker maskSample=new ProfilerMarker("CG.SharedOutline.Mask");
    static readonly ProfilerMarker outputSample=new ProfilerMarker("CG.SharedOutline.Output");
    static readonly ProfilerMarker collectSample=new ProfilerMarker("CG.SharedOutline.Collect");
    static readonly ProfilerMarker sortSample=new ProfilerMarker("CG.SharedOutline.Sort");
    static readonly ProfilerMarker recordSample=new ProfilerMarker("CG.SharedOutline.Record");
    static readonly ProfilerMarker executeSample=new ProfilerMarker("CG.SharedOutline.Execute");
    static readonly int MaskTextureId=Shader.PropertyToID("_CGSharedOutlineMask"),LayoutId=Shader.PropertyToID("_CGSharedOutlineLayout"),
        ViewProjectionId=Shader.PropertyToID("_CGSharedOutlineVP"),ThresholdId=Shader.PropertyToID("_CGSharedOutlineThreshold"),
        MaskEntityId=Shader.PropertyToID("_CGMaskEntity"),OutputEntityId=Shader.PropertyToID("_CGOutlineEntity"),
        ColorId=Shader.PropertyToID("_Color"),WidthId=Shader.PropertyToID("_Width");

    // Adopted 2026-10-24 (user decision after step 2/3 comparison). SetEnabled(false) restores the original per-body path.
    public static bool Enabled {get;private set;}=true;
    public static void SetEnabled(bool value)=>Enabled=value;

    struct Part
    {
        public SpriteRenderer renderer;
        public int layer,order,entity,sequence;
        public float z;
    }
    struct Entry
    {
        public MeshRenderer output;
        public Bounds bounds;
        public int frontLayerId,frontOrder;
    }
    sealed class PartOrder : IComparer<Part>
    {
        // Back to front like the 2D transparent queue: sorting layer, order, then farther z first.
        public int Compare(Part a,Part b)
        {
            if(a.layer!=b.layer)return a.layer<b.layer?-1:1;
            if(a.order!=b.order)return a.order<b.order?-1:1;
            if(a.z!=b.z)return a.z>b.z?-1:1;
            return a.sequence.CompareTo(b.sequence);
        }
    }
    static readonly PartOrder partOrder=new PartOrder();

    readonly List<Part> parts=new List<Part>();
    readonly List<Entry> entries=new List<Entry>();
    readonly List<Material> entityMaterials=new List<Material>();
    readonly Dictionary<int,int> layerValues=new Dictionary<int,int>();
    int LayerValue(int id){if(!layerValues.TryGetValue(id,out var v)){v=SortingLayer.GetLayerValueFromID(id);layerValues.Add(id,v);}return v;}
    MaterialPropertyBlock outputProperties;
    Material maskMaterial,outputMaterial;
    Mesh quad;
    CommandBuffer commands;
    RenderTexture mask;
    Camera camera;
    Transform parent;
    Color color;
    int radius;
    float threshold;
    bool active;

    public int LastEntityCount {get;private set;}
    public int LastPartCount {get;private set;}

    // Only the gameplay camera setup that the quad placement below assumes.
    public static bool Supports(Camera camera,Transform parent)
    {
        if(camera==null||!camera.orthographic||camera.stereoEnabled||camera.allowDynamicResolution)return false;
        if(!camera.transform.rotation.Equals(Quaternion.identity))return false;
        var viewport=camera.pixelRect;if(viewport.x!=0||viewport.y!=0||viewport.width<=0||viewport.height<=0)return false;
        if(parent!=null&&!parent.rotation.Equals(Quaternion.identity))return false;
        return SystemInfo.SupportsRenderTextureFormat(RenderTextureFormat.RFloat);
    }

    public void Begin(Camera camera,Transform parent,int radius,Color color,float threshold)
    {
        this.camera=camera;this.parent=parent;this.radius=Mathf.Clamp(radius,0,8);this.color=color;this.threshold=Mathf.Clamp(threshold,.01f,1);
        parts.Clear();entries.Clear();active=camera!=null&&this.radius>0&&color.a>0;
    }

    // Returns false when the body has no visible source part; its output is hidden.
    public bool Add(GameObject host,ref MeshRenderer output,List<SpriteRenderer> sources)
    {
        if(!active||host==null||sources==null){Hide(output);return false;}
        using var collectScope=collectSample.Auto();
        int entity=entries.Count+1,first=parts.Count;
        SpriteRenderer front=null;int frontLayer=0;Bounds bounds=default;
        foreach(var r in sources)
        {
            // Same source filter as CoreGuardianSelectionOutline.PresentMask. The emphasis owner builds `sources`
            // from activeInHierarchy renderers in this same call, so that check is not repeated here.
            if(r==null||!r.enabled||r.sprite==null||r.color.a<threshold)continue;
            int layer=LayerValue(r.sortingLayerID);var rb=r.bounds;
            if(front==null){front=r;frontLayer=layer;bounds=rb;}
            else
            {
                bounds.Encapsulate(rb);
                // >= keeps the original stable OrderBy(...).ThenBy(...).Last() tie behaviour.
                if(layer>frontLayer||(layer==frontLayer&&r.sortingOrder>=front.sortingOrder)){front=r;frontLayer=layer;}
            }
            parts.Add(new Part{renderer=r,layer=layer,order=r.sortingOrder,entity=entity,sequence=parts.Count,z=rb.center.z});
        }
        if(front==null){Hide(output);return false;}
        EnsureResources();
        if(output==null)
        {
            // Own child: never touches the host or the original SelectionOutline output transform.
            var child=new GameObject("SharedOutline"){hideFlags=HideFlags.DontSave};child.transform.SetParent(host.transform,false);
            child.AddComponent<MeshFilter>().sharedMesh=quad;
            output=child.AddComponent<MeshRenderer>();
            output.sharedMaterial=outputMaterial;output.shadowCastingMode=ShadowCastingMode.Off;output.receiveShadows=false;
            output.lightProbeUsage=LightProbeUsage.Off;output.reflectionProbeUsage=ReflectionProbeUsage.Off;
        }
        entries.Add(new Entry{output=output,bounds=bounds,frontLayerId=front.sortingLayerID,frontOrder=front.sortingOrder});
        return true;
    }

    public static void Hide(MeshRenderer output){if(output!=null)output.enabled=false;}

    public void End()
    {
        LastEntityCount=entries.Count;LastPartCount=parts.Count;
        if(!active||entries.Count==0)return;
        var viewport=camera.pixelRect;int width=(int)viewport.width,height=(int)viewport.height,pad=radius+1;
        using(maskSample.Auto())
        {
            using(sortSample.Auto())parts.Sort(partOrder);
            EnsureMask(width+pad*2,height+pad*2);
            // Expand the camera projection by `pad` pixels on every side so off-screen neighbours stay in the mask.
            float ax=-pad/(float)width*2-1,bx=(width+pad)/(float)width*2-1,ay=-pad/(float)height*2-1,by=(height+pad)/(float)height*2-1;
            var crop=Matrix4x4.identity;crop.m00=2/(bx-ax);crop.m11=2/(by-ay);crop.m03=-(bx+ax)/(bx-ax);crop.m13=-(by+ay)/(by-ay);
            var vp=GL.GetGPUProjectionMatrix(crop*camera.projectionMatrix,true)*camera.worldToCameraMatrix;
            commands.Clear();commands.SetRenderTarget(mask);commands.ClearRenderTarget(false,true,Color.clear);
            commands.SetGlobalMatrix(ViewProjectionId,vp);commands.SetGlobalFloat(ThresholdId,threshold);
            using(recordSample.Auto())
            {
                // One immutable material per entity id: no per-entity global writes inside the command buffer.
                while(entityMaterials.Count<entries.Count)
                {
                    var m=new Material(maskMaterial){name="Shared outline mask "+(entityMaterials.Count+1),hideFlags=HideFlags.DontSave};
                    m.SetFloat(MaskEntityId,entityMaterials.Count+1);entityMaterials.Add(m);
                }
                foreach(var part in parts)commands.DrawRenderer(part.renderer,entityMaterials[part.entity-1],0,0);
            }
            using(executeSample.Auto())Graphics.ExecuteCommandBuffer(commands);
            Shader.SetGlobalTexture(MaskTextureId,mask);
            Shader.SetGlobalVector(LayoutId,new Vector4(pad,pad,width,height));
        }
        using(outputSample.Auto())
        {
            float pixel=camera.orthographicSize*2/height;
            float margin=(radius+2)*pixel*2;
            for(int i=0;i<entries.Count;i++)
            {
                var entry=entries[i];var output=entry.output;var t=output.transform;
                var size=entry.bounds.size;var parentScale=t.parent!=null?t.parent.lossyScale:Vector3.one;
                t.position=entry.bounds.center;
                t.localScale=new Vector3((size.x+margin)/parentScale.x,(size.y+margin)/parentScale.y,1);
                output.sortingLayerID=entry.frontLayerId;output.sortingOrder=entry.frontOrder+3;
                outputProperties.Clear();outputProperties.SetFloat(OutputEntityId,i+1);
                outputProperties.SetColor(ColorId,color);outputProperties.SetInt(WidthId,radius);
                output.SetPropertyBlock(outputProperties);
                output.enabled=true;
            }
        }
    }

    void EnsureResources()
    {
        if(maskMaterial==null)maskMaterial=new Material(Resources.Load<Shader>("CoreGuardian/SharedOutlineMask")){name="Shared outline mask",hideFlags=HideFlags.DontSave};
        if(outputMaterial==null)outputMaterial=new Material(Resources.Load<Shader>("CoreGuardian/SharedOutlineOutput")){name="Shared outline output",hideFlags=HideFlags.DontSave};
        if(outputProperties==null)outputProperties=new MaterialPropertyBlock();
        if(commands==null)commands=new CommandBuffer{name="Shared emphasis outline mask"};
        if(quad==null)
        {
            quad=new Mesh{name="Shared outline quad",hideFlags=HideFlags.DontSave};
            quad.vertices=new[]{new Vector3(-.5f,-.5f,0),new Vector3(.5f,-.5f,0),new Vector3(.5f,.5f,0),new Vector3(-.5f,.5f,0)};
            quad.triangles=new[]{0,1,2,0,2,3};quad.RecalculateBounds();
        }
    }
    void EnsureMask(int width,int height)
    {
        if(mask!=null&&mask.width==width&&mask.height==height)return;
        ReleaseMask();
        mask=new RenderTexture(width,height,0,RenderTextureFormat.RFloat,RenderTextureReadWrite.Linear)
            {name="Shared emphasis outline mask",filterMode=FilterMode.Point,wrapMode=TextureWrapMode.Clamp,useMipMap=false,autoGenerateMips=false,hideFlags=HideFlags.DontSave};
        mask.Create();
    }
    void ReleaseMask(){if(mask==null)return;mask.Release();Dispose(mask);mask=null;}

    public void Release()
    {
        parts.Clear();entries.Clear();active=false;
        ReleaseMask();commands?.Dispose();commands=null;
        foreach(var m in entityMaterials)Dispose(m);entityMaterials.Clear();layerValues.Clear();
        Dispose(maskMaterial);maskMaterial=null;Dispose(outputMaterial);outputMaterial=null;Dispose(quad);quad=null;
    }
    static void Dispose(Object value){if(value==null)return;if(Application.isPlaying)Object.Destroy(value);else Object.DestroyImmediate(value);}
}
