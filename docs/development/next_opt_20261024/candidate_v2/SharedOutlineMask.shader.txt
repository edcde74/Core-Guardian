Shader "CoreGuardian/SharedOutlineMask"
{
    // Prototype (default OFF): writes the owning emphasis entity id of each visible body pixel into one screen-size mask.
    Properties { _MainTex("Sprite",2D)="white"{} _CGMaskEntity("Entity id",Float)=0 }
    SubShader
    {
        Tags {"RenderPipeline"="UniversalPipeline"}
        Pass
        {
            Cull Off ZWrite Off ZTest Always
            Blend Off
            HLSLPROGRAM
            #pragma vertex Vert
            #pragma fragment Frag
            #pragma target 3.5
            #include "Packages/com.unity.render-pipelines.universal/ShaderLibrary/Core.hlsl"
            TEXTURE2D(_MainTex);SAMPLER(sampler_MainTex);
            float4x4 _CGSharedOutlineVP;
            float _CGSharedOutlineThreshold,_CGMaskEntity;
            struct Varyings {float4 positionCS:SV_POSITION;float2 uv:TEXCOORD0;nointerpolation float alpha:TEXCOORD1;};
            Varyings Vert(float4 positionOS:POSITION,float2 uv:TEXCOORD0,float4 color:COLOR)
            {
                // SpriteRenderer flip is applied by the sprite shader, not by the transform.
                float3 local=float3(positionOS.xy*unity_SpriteProps.xy,positionOS.z);
                Varyings o;o.positionCS=mul(_CGSharedOutlineVP,mul(unity_ObjectToWorld,float4(local,1)));o.uv=uv;
                // Renderer alpha may arrive as vertex colour or as unity_SpriteColor depending on the draw path.
                o.alpha=color.a*unity_SpriteColor.a;return o;
            }
            float4 Frag(Varyings i):SV_Target
            {
                // Same per-pixel rule as SelectionMask: sprite alpha x renderer alpha >= threshold.
                clip(SAMPLE_TEXTURE2D(_MainTex,sampler_MainTex,i.uv).a*i.alpha-_CGSharedOutlineThreshold);
                return float4(_CGMaskEntity,0,0,1);
            }
            ENDHLSL
        }
    }
}
